{\footnotesize
\[ \varepsilon_{|Z|} \;\approx\; \underbrace{\frac{R_f/|Z|}{A_{\mathrm{eff}}}}_{\text{systematic}} \;+\; \underbrace{\frac{i_n\,|Z|}{V_1}\sqrt{\frac{2}{N M}}}_{\text{random}} \]}

\begin{frame}{Calibrating impedance: what cancels, and what a 20\,\% poly costs}
\centering
{\footnotesize
$\hat Z = Z_{\mathrm{cal}}\,\dfrac{(\hat V_{\mathrm{P}}/\hat V_{\mathrm{OUT}})}{(\hat V_{\mathrm{P}}/\hat V_{\mathrm{OUT}})_{\mathrm{cal}}}$ \quad same gain code, same frequency, same converter\par}

{\footnotesize
\begin{tabular}{@{}l>{\raggedright\arraybackslash}p{4.6cm}>{\raggedright\arraybackslash}p{4.6cm}@{}}
\toprule
 & On-chip $Z_{\mathrm{cal}}$ (per decade, muxed in) & Off-chip standard (at test) \\
\midrule
Cancels     & $R_f$, its pole, the DAC, the LSB, A2's gain floor & the same \\
Value known & to the poly's chip-to-chip spread, $\sim$20\,\% & to the standard, $<0.1$\,\% \\
When        & any time, in the field, before every point & once per chip, stored in flash \\
Tracks      & temperature and ageing, since $Z_{\mathrm{cal}}$ and $R_f$ are the same poly & the channel on the day it was tested \\

\bottomrule
\end{tabular}}
