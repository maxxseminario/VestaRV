\begin{tikzpicture}[
  call/.style={font=\scriptsize, align=left, fill=white, fill opacity=0.92,

  lead/.style={draw=black!55, line width=0.6pt, -{Latex[length=1.8mm]}},

  \node[inner sep=0] (S) {\trmfit[20cm]{11.4cm}{4.7cm}{\MaestroRoot sch_tb_pixel.tex}};
  % \schat{fx}{fy}{name}: a coordinate at fractions of the schematic box
  \newcommand{\schat}[3]{%
    \path ($(S.south west)!#1!(S.south east)$) coordinate (tmpx);
    \path ($(S.south west)!#2!(S.north west)$) coordinate (tmpy);
    \path (tmpx |- tmpy) coordinate (#3);}

  \schat{0.22}{0.90}{Pre}      % IPRB1 on the RE wire

  \schat{0.905}{0.58}{Pout}    % V_OUT node

  \begin{scope}[shift={($(S.north west)+(0.2,0.55)$)}]
    \draw[black!40, line width=0.3pt] (0,0) -- (2.4,0);

  % ---- callouts on the loop --------------------------------------------------
  \schat{0.36}{0.905}{Prewire}
