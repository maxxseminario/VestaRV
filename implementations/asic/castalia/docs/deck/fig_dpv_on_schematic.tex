%% HAND-DRAWN for this deck.  The TRM's potentiostat schematic (sch_tb_pixel,
%% input unmodified through \trmfit) with the differential-pulse voltammetry
%% program sketched at the potential source and the resolved peaks sketched
%% at the output.  Sketches only; the simulated waveforms are the next slide
%% (fig_electrode_dpv).  Programme per TRM note_electrode_dpv: 4 mV treads,
%% 50 mV pulses, current sampled 5 ms before each edge (S1 before the pulse,
%% S2 at its end), dI = I(S2) - I(S1).

\begin{tikzpicture}[
  wv/.style={line width=1.0pt},
  lab/.style={font=\scriptsize, text=black!70}]

  \node[inner sep=0] (S) {\trmfit[20cm]{11.4cm}{3.3cm}{\MaestroRoot sch_tb_pixel.tex}};

  % ---- excitation: staircase with a pulse on every tread, above the source
  \begin{scope}[shift={($(S.north west)+(0.2,0.35)$)}]
    \draw[black!40, line width=0.3pt, -{Latex[length=1.2mm]}] (0,-0.3) -- (2.75,-0.3) node[lab, anchor=west, inner sep=1pt] {$t$};
    \draw[black!40, line width=0.3pt, -{Latex[length=1.2mm]}] (0,-0.3) -- (0,0.42) node[lab, anchor=south, inner sep=1pt] {$v_{\mathrm{step}}$};
    \draw[wv, mtxA]
      (0.0,-0.30) -- (0.3,-0.30) -- (0.3,-0.05) -- (0.45,-0.05) -- (0.45,-0.22)
      -- (0.75,-0.22) -- (0.75, 0.03) -- (0.9, 0.03) -- (0.9,-0.14)
      -- (1.2,-0.14) -- (1.2, 0.11) -- (1.35,0.11) -- (1.35,-0.06)
      -- (1.65,-0.06) -- (1.65,0.19) -- (1.8,0.19) -- (1.8,0.02)
      -- (2.1,0.02) -- (2.1,0.27) -- (2.25,0.27) -- (2.25,0.10) -- (2.6,0.10);
    \fill[black!70] (1.18,-0.14) circle (0.035);
    \fill[black!70] (1.33, 0.11) circle (0.035);
    \node[lab, anchor=north east] at (1.16,-0.17) {$S_1$};
    \node[lab, anchor=south west] at (1.35, 0.13) {$S_2$};
    \node[lab, anchor=west, align=left] at (3.0,0.05) {4\,mV treads,\\50\,mV pulses};
  \end{scope}

  % ---- response in time: a charging spike on every edge, decaying to the
  %      faradaic level; the two samples sit where the charging has died away
  \begin{scope}[shift={($(S.north east)+(-3.3,0.35)$)}]
    \draw[black!40, line width=0.3pt, -{Latex[length=1.2mm]}] (0,-0.3) -- (2.75,-0.3) node[lab, anchor=west, inner sep=1pt] {$t$};
    \draw[black!40, line width=0.3pt, -{Latex[length=1.2mm]}] (0,-0.3) -- (0,0.42) node[lab, anchor=south, inner sep=1pt] {$v_{\mathrm{out}}$};
    % one tread = 0.45 wide: pulse on for 0.15, off for 0.30; baseline drifts up
    \foreach \k/\b in {0/-0.26, 1/-0.20, 2/-0.13, 3/-0.05, 4/0.04}{
      \draw[wv, mtxB] plot[samples=30, domain=0:0.15]
        ({0.3+\k*0.45+\x}, {\b+0.10 + 0.24*exp(-\x/0.035)});
      \draw[wv, mtxB] plot[samples=30, domain=0:0.30]
        ({0.45+\k*0.45+\x}, {\b+0.02 - 0.18*exp(-\x/0.035)});
      \draw[wv, mtxB] (0.3+\k*0.45, \b+0.02) -- (0.3+\k*0.45, \b+0.34);
      \draw[wv, mtxB] (0.45+\k*0.45, \b+0.10) -- (0.45+\k*0.45, \b-0.16);
    }
    \fill[black!70] (1.19,-0.11) circle (0.035);
    \fill[black!70] (1.34,-0.03) circle (0.035);
    \node[lab, anchor=north east] at (1.17,-0.14) {$S_1$};
    \node[lab, anchor=south west] at (1.36,-0.01) {$S_2$};
  \end{scope}

  % ---- post-processing, to the right of the die: the voltammogram
  \begin{scope}[shift={($(S.east)+(1.1,-0.1)$)}]
    \draw[black!40, line width=0.3pt, -{Latex[length=1.2mm]}] (0,-0.3) -- (2.75,-0.3) node[lab, anchor=west, inner sep=1pt] {$E$};
    \draw[black!40, line width=0.3pt, -{Latex[length=1.2mm]}] (0,-0.3) -- (0,0.42) node[lab, anchor=south, inner sep=1pt] {$\Delta I$};
    \foreach \c/\px/\h in {mtxA/0.45/0.55, mtxB/1.05/0.42, mtxC/1.6/0.30, mtxD/2.15/0.22}{
      \draw[wv, \c] plot[smooth, samples=40, domain=\px-0.32:\px+0.32]
        (\x, {-0.3 + \h*exp(-((\x-\px)/0.11)^2)});
    }
    \node[lab, anchor=north, align=center] at (1.35,-0.45) {firmware, per tread:\\$\Delta I = I(S_2)-I(S_1)$ against $E$\\one peak per marker};
  \end{scope}
  \draw[draw=black!55, line width=0.6pt, -{Latex[length=1.8mm]}]
       ($(S.east)+(0.15,0)$) -- ($(S.east)+(0.95,0)$);
\end{tikzpicture}
